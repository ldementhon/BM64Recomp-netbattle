cmake_minimum_required(VERSION 3.20)
project(bm64_lobby LANGUAGES C CXX)
if(WIN32)
    add_compile_definitions(NOMINMAX)
endif()
include(../cmake/Enet.cmake)

add_executable(bm64-lobby main.cpp)
target_link_libraries(bm64-lobby PRIVATE enet)
target_compile_features(bm64-lobby PRIVATE cxx_std_20)
if(NOT MSVC)
    target_compile_options(bm64-lobby PRIVATE -Wall -Wextra -Werror)
endif()
include(CTest)
if(BUILD_TESTING)
    set(lobby_tests lobby_room lobby_network server_clock input_history input_network match_channel game_session lobby_client lobby_preferences release_file)
    if(APPLE OR WIN32)
        list(APPEND lobby_tests fingerprint lobby_worker)
    endif()
    foreach(test ${lobby_tests})
        add_executable(${test} ../tests/${test}.cpp)
        target_compile_features(${test} PRIVATE cxx_std_20)
        target_link_libraries(${test} PRIVATE enet)
        if(MSVC)
            target_compile_options(${test} PRIVATE /UNDEBUG)
        else()
            target_compile_options(${test} PRIVATE -UNDEBUG -Wall -Wextra -Werror)
        endif()
        add_test(NAME ${test} COMMAND ${test})
        set_tests_properties(${test} PROPERTIES TIMEOUT 30)
    endforeach()
    if(WIN32)
        target_link_libraries(fingerprint PRIVATE bcrypt)
        target_link_libraries(lobby_worker PRIVATE bcrypt)
    endif()
    target_include_directories(lobby_preferences PRIVATE ../lib/N64ModernRuntime/thirdparty)
    target_sources(game_session PRIVATE ../src/online/game_session.cpp)
    target_sources(lobby_client PRIVATE ../src/online/game_session.cpp)
    if(UNIX)
        add_executable(network_game ../tests/network_game.cpp ../src/online/game_session.cpp)
        target_compile_features(network_game PRIVATE cxx_std_20)
        target_link_libraries(network_game PRIVATE enet)
        target_compile_options(network_game PRIVATE -Wall -Wextra -Werror)
    endif()
    if(APPLE OR WIN32)
        target_sources(lobby_worker PRIVATE ../src/online/game_session.cpp)
        # One scripted player for cross-machine matches; not a CTest.
        add_executable(network_peer ../tests/network_peer.cpp ../src/online/game_session.cpp)
        target_compile_features(network_peer PRIVATE cxx_std_20)
        target_link_libraries(network_peer PRIVATE enet)
        if(WIN32)
            target_link_libraries(network_peer PRIVATE bcrypt)
        else()
            target_compile_options(network_peer PRIVATE -Wall -Wextra -Werror)
        endif()
    endif()
endif()
